1  REM ======================================
2  REM  JOYPORT TEST PROGRAM
3  REM  BASED ON THE TEST PROGRAM IN THE
4  REM  SIRIUS JOYPORT MANUAL, WITH A MENU,
5  REM  RETRIES AND 40-COLUMN PROMPTS.
6  REM
7  REM  AN0 ($C058/$C059) PICKS THE JACK WHEN
8  REM  THE REAR SWITCH IS IN THE MIDDLE.
9  REM  AN1 ($C05A/$C05B) PICKS LEFT/RIGHT OR
10  REM  UP/DOWN. AN ATARI SWITCH READS BELOW
11  REM  128 WHEN CLOSED; A PADDLE BUTTON READS
12  REM  ABOVE 127 WHEN PRESSED.
13  REM ======================================
90  POKE 768,0
100  TEXT : HOME : GOSUB 950:SK = 0
110  PRINT "JOYPORT TEST PROGRAM": PRINT
120  PRINT "PLUG TWO ATARI JOYSTICKS, OR TWO SETS"
130  PRINT "OF GAME PADDLES, INTO THE JOYPORT.": PRINT
140  PRINT "1  ATARI JOYSTICK, REAR SWITCH RIGHT"
150  PRINT "2  ATARI JOYSTICK, REAR SWITCH LEFT"
160  PRINT "3  BOTH ATARI JOYSTICKS, REAR SWITCH"
170  PRINT "   IN THE MIDDLE"
180  PRINT "4  GAME PADDLES"
190  PRINT "5  EVERY TEST, ONE AFTER ANOTHER"
195  PRINT "B  BEEPS: ";: IF  PEEK (768) THEN  PRINT "ON": GOTO 200
197  PRINT "OFF"
200  PRINT "Q  QUIT": PRINT
210  PRINT "CHOOSE A TEST: ";
220  GET A$
230  IF A$ = "1" THEN  PRINT A$: GOSUB 1000: GOTO 800
240  IF A$ = "2" THEN  PRINT A$: GOSUB 1100: GOTO 800
250  IF A$ = "3" THEN  PRINT A$: GOSUB 1200: GOTO 800
260  IF A$ = "4" THEN  PRINT A$: GOSUB 1500: GOTO 800
270  IF A$ = "5" THEN  PRINT A$: GOSUB 1000: GOSUB 1100: GOSUB 1200: GOSUB 1500: GOTO 800
280  IF A$ = "Q" OR A$ =  CHR$ (113) THEN  HOME : END
285  IF A$ = "B" OR A$ =  CHR$ (98) THEN  POKE 768,1 -  PEEK (768): GOTO 100
290  GOTO 220
800  REM A TEST FINISHED. SK IS HOW MANY STEPS WERE SKIPPED
810  GOSUB 950: HOME : GOSUB 990: IF SK > 0 THEN 860
820  PRINT "ALL DONE. EVERY STEP OF THAT TEST": PRINT "PASSED.": PRINT
830  PRINT "PRESS SPACE FOR THE MENU."
840  GET A$: IF A$ <  > " " THEN 840
850  GOTO 100
860  PRINT "DONE, WITH ";SK;" STEP";: IF SK > 1 THEN  PRINT "S";
870  PRINT " SKIPPED.": PRINT "EVERY OTHER STEP PASSED.": PRINT : GOTO 830
900  REM ESC: BACK TO THE MENU
910  GOSUB 950: CLEAR : GOTO 100
950  REM BOTH ANNUNCIATORS OFF, KEYBOARD CLEAR
960  POKE  - 16296,0: POKE  - 16294,0: POKE  - 16368,0: RETURN
980  REM BEEP IF BEEPS ARE ON. THE SETTING IS KEPT AT 768 ($300), NOT IN A VARIABLE, BECAUSE ESC CLEARS THE VARIABLES
990  IF  PEEK (768) THEN  PRINT  CHR$ (7);
995  RETURN
1000  REM ATARI JOYSTICK, REAR SWITCH RIGHT
1010 T$ = "ATARI JOYSTICK TEST"
1020 F$ = "SET THE FRONT SWITCH TO THE FRONT"
1030 R$ = "AND THE REAR SWITCH TO THE RIGHT."
1040  GOSUB 1900: POKE  - 16296,0:J$ = "RIGHT": GOSUB 1300: RETURN
1100  REM ATARI JOYSTICK, REAR SWITCH LEFT
1110 T$ = "ATARI JOYSTICK TEST"
1120 F$ = "SET THE FRONT SWITCH TO THE FRONT"
1130 R$ = "AND THE REAR SWITCH TO THE LEFT."
1140  GOSUB 1900: POKE  - 16296,0:J$ = "LEFT": GOSUB 1300: RETURN
1200  REM BOTH ATARI JOYSTICKS, REAR SWITCH IN THE MIDDLE
1210 T$ = "TWO ATARI JOYSTICKS TEST"
1220 F$ = "SET THE FRONT SWITCH TO THE FRONT"
1230 R$ = "AND THE REAR SWITCH TO THE MIDDLE."
1240  GOSUB 1900
1250  POKE  - 16296,0:J$ = "LEFT": GOSUB 1300
1260  POKE  - 16295,0:J$ = "RIGHT": GOSUB 1300
1270  POKE  - 16296,0: RETURN
1300  REM ONE ATARI JOYSTICK. J$ IS WHICH ONE
1310 H = 0: POKE  - 16294,0
1320 S =  - 16285:P$ = J$ + " JOYSTICK: PUSH IT RIGHT": GOSUB 2000
1330 S =  - 16286:P$ = J$ + " JOYSTICK: PUSH IT LEFT": GOSUB 2000
1340 S =  - 16287:P$ = J$ + " JOYSTICK: PRESS FIRE": GOSUB 2000
1350  POKE  - 16293,0
1360 S =  - 16286:P$ = J$ + " JOYSTICK: PUSH IT UP": GOSUB 2000
1370 S =  - 16285:P$ = J$ + " JOYSTICK: PUSH IT DOWN": GOSUB 2000
1380  POKE  - 16294,0: RETURN
1500  REM GAME PADDLES, IN EACH REAR SWITCH POSITION
1510 T$ = "GAME PADDLE TEST"
1520 F$ = "SET THE FRONT SWITCH TO THE REAR"
1530 R$ = "AND THE REAR SWITCH TO THE RIGHT."
1540  GOSUB 1900: GOSUB 1600
1550 R$ = "AND THE REAR SWITCH TO THE LEFT."
1560  GOSUB 1900: GOSUB 1600
1570 R$ = "AND THE REAR SWITCH TO THE MIDDLE."
1580  GOSUB 1900: GOSUB 1600
1590  POKE  - 16295,0:Y = 2:BT =  - 16287: GOSUB 4000:Y = 3:BT =  - 16286: GOSUB 4000: POKE  - 16296,0: RETURN
1600  REM PADDLES 0 AND 1 WITH THEIR BUTTONS
1610  POKE  - 16296,0:Y = 0:BT =  - 16287: GOSUB 4000:Y = 1:BT =  - 16286: GOSUB 4000: RETURN
1900  REM SHOW T$, F$ AND R$, THEN WAIT FOR SPACE
1910  HOME : PRINT T$: PRINT : PRINT F$: PRINT R$: PRINT
1920  PRINT "PRESS SPACE WHEN READY, ESC FOR MENU."
1930  GET A$: IF A$ =  CHR$ (27) THEN 900
1940  IF A$ <  > " " THEN 1930
1950  RETURN
2000  REM WAIT FOR THE SWITCH AT S. H IS 1 WHEN IT READS HIGH PRESSED
2010  HOME : PRINT P$: VTAB 22: PRINT "S: SKIP THIS STEP   ESC: MENU": POKE  - 16368,0
2020  FOR X = 1 TO 1500
2030  IF ( PEEK (S) > 127) = H THEN  GOSUB 990: RETURN
2040 K =  PEEK ( - 16384): IF K = 155 THEN 900
2045  IF K = 211 OR K = 243 THEN  POKE  - 16368,0:SK = SK + 1: RETURN
2050  NEXT X
2060  GOSUB 3000: IF SF THEN SF = 0:SK = SK + 1: RETURN
2070  GOTO 2010
3000  REM NOTHING HAPPENED: RETRY THIS STEP, SKIP IT, OR GO TO THE MENU. SF IS 1 TO SKIP
3010  VTAB 4: GOSUB 990: PRINT "NOTHING HAPPENED. CHECK THE SWITCHES"
3020  PRINT "AND THE CONNECTIONS.": PRINT
3030  PRINT "SPACE: TRY THIS STEP AGAIN"
3040  PRINT "S:     SKIP THIS STEP": PRINT "ESC:   BACK TO THE MENU"
3050  POKE  - 16368,0
3060  GET A$: IF A$ =  CHR$ (27) THEN 900
3070  IF A$ = "S" OR A$ =  CHR$ (115) THEN SF = 1: RETURN
3075  IF A$ <  > " " THEN 3060
3080  RETURN
4000  REM PADDLE Y, THEN ITS BUTTON AT BT. SKIPPING THE PADDLE SKIPS ITS BUTTON
4010  HOME : PRINT "TURN PADDLE ";Y;" ALL THE WAY LEFT,"
4020  PRINT "THEN ALL THE WAY RIGHT."
4030  VTAB 22: PRINT "S: SKIP THIS PADDLE   ESC: MENU": POKE  - 16368,0
4040 L1 = 0:H1 = 0
4050  FOR X = 1 TO 600
4060 V =  PDL (Y): VTAB 4: HTAB 1: PRINT "PADDLE ";Y;" READS ";V;"   "
4070  IF V < 64 THEN L1 = 1
4080  IF V > 191 THEN H1 = 1
4090  IF L1 AND H1 THEN 4130
4100 K =  PEEK ( - 16384): IF K = 155 THEN 900
4105  IF K = 211 OR K = 243 THEN  POKE  - 16368,0:SK = SK + 1: RETURN
4110  NEXT X
4120  GOSUB 3000: IF SF THEN SF = 0:SK = SK + 1: RETURN
4125  GOTO 4010
4130  GOSUB 990:S = BT:H = 1:P$ = "PADDLE " +  STR$ (Y) + ": PRESS ITS BUTTON": GOSUB 2000: RETURN
